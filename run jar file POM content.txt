<!-- Added the class name of jar file (which class to run, in our case it is App class) -->
<configuration>
          <archive>
            <manifest>
              <mainClass>demo.project.App</mainClass>
            </manifest>
          </archive>
</configuration> 

OR

<build>
    <plugins>
        <plugin>
            <groupId>org.apache.maven.plugins</groupId>
            <artifactId>maven-jar-plugin</artifactId>
            <version>3.2.0</version>
            <configuration>
                <archive>
                    <manifest>
                        <mainClass>com.App</mainClass>
                    </manifest>
                </archive>
            </configuration>
        </plugin>
    </plugins>
</build>
